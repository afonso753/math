\textbf{Seja $f: G -> G'$ um homomorfismo de grupos. Supondo que $|G|=18$, $|G'|=15$ e que f não é o homomorfismo trivial, determine $|ker f|$. Justifique a resposta.}\\

\underline{Vejamos que $ker f < G$:}\\
$ker f$=$\{x \in G \mid f(x)=e_G'\}$
\begin{itemize}
    \item $ker f \subseteq G$;
    \item $f(e_G)=e_G'\rightarrow ker f \neq \emptyset $;
    \item $\forall{x,y \in ker f}, f(xy^{-1})=f(x)f(y^{-1})=e_G'{f(y)}^{-1}=(e_G')^{-1}=e_G' \rightarrow xy^{-1} \in ker f$.
Portanto, $ker f < G$.
\end{itemize}


Como $ker f < G$ temos, pelo teorema de $Lagrange$,
$$18=|G|=[G:ker f]|ker f|=|im f||ker f| \;\; \; \; \; \; \;\; \; \;   (1) $$ Ou seja $|im f|$ divide $|G| \leftrightarrow |im f| \in \{1,2,3,6,9,18\} $ \\


\underline{Vejamos que $im f < G'$:}\\
$im f$=$\{x \in G' \mid x=f(a), \forall{a \in G}\}$
\begin{itemize}
    \item $im f \subseteq G'$;
    \item Sejam $f(x),f(y) \in im f \rightarrow f(x)f(y)=f(xy) \in im f$ (fechado para multiplicação);
    \item $f(e_G)=e_G'$ (fechado para o elemento neutro);
    \item Seja $f(x) \in im f \rightarrow (f(x))^{-1}=f(x^{-1}) \in im f$ (fechado para os inversos).
Portanto, $im f < G'$.
\end{itemize}



Por outro lado, como $im f < G'$ temos, outra vez, pelo teorema de $Lagrange$, 
$$15=|G'|=[G':im f]|im f|\;\; \; \; \; \; \;\; \; \;   (2)$$ Ou seja $|im f|$ divide $|G'| \leftrightarrow |im f| \in \{1,3,5\}$\\
Juntando (1) e (2) temos $|im f| \in \{1,3\}$. Como $f$ não é o homomorfismo trivial, então $|im f| \neq1 \leftrightarrow |im f|=3$.\\
De (1) vem que $$|ker f|=\frac{|G|}{|im f|}= \frac{18}{3}=6$$
