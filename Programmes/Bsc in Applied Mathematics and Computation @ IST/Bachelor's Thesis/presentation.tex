\begin{frame}{}
\begin{block}{Definição}
    Dado $H \in (0,1)$, um movimento Browniano fracionário de parametro $H$ é um processo Gaussiano centrado de função de covariancia
    $$\mathbb{E}[B^H_tB^H_s]= \frac{1}{2}(t^{2H}+s^{2H}-|t-s|^{2H})$$
\end{block}

\pause

\vspace{0.5cm}

Se $H=\frac{1}{2}$ recuperamos o movimento Browniano clássico
$$\mathbb{E}[B^{1/2}_tB^{1/2}_s]= \frac{1}{2}(t+s-|t-s|)=\min (s,t)$$

\end{frame}

\begin{frame}{Incrementos}
    \begin{block}{Teorema}
        O movimento Browniano fracionário tem incrementos estacionários
        $$B_{t+s}^H-B_t^H \overset{\mathcal{D}}{=}B_s^H \;\;\;\text{ para quaisquer } t,s \geq 0$$
    \end{block}
    
    \vspace{0.5cm}

    \pause

    \begin{block}{Teorema}
        
    Dados $0<s_1<t_1<s_2<t_2$

    \vspace{0.3cm}
       
    $\mathbb{E}[(B_{t_1}^H-B_{s_1}^H)(B_{t_2}^H-B_{s_2}^H)] 
    \begin{cases}
        >0,\;\; H>1/2\;\; \text{(persistentes)}\\
        =0,\;\; H=1/2\;\;\text{(independenets)}\\
        <0,\;\; H<1/2\;\; \text{(antipersistentes)}\\
    \end{cases}$

    \end{block}

\end{frame}

\begin{frame}{Martingale}
Fixe-se um espaço de probabilidade filtrado $(\Omega, \mathcal{F}, (\mathcal{F}_{n})_{n \in \mathbb{N}}, \mathbb{P})$
\begin{block}{Definição}
    Seja $(X_t)_{t\geq0}$ um processo estocástico adaptado à filtração $(\mathcal{F}_t)_{t\geq0}$. Dizemos que $(X_t)_{t \geq 0}$ é uma martingale se
    \vspace{0.2cm}
    \begin{enumerate}
        \item $\mathbb{E}[|X_t|]<\infty$, para qualquer t
        \item $\mathbb{E}[X_t |\mathcal{F}_s]=X_s$, para quaisquer $s<t$
    \end{enumerate}
\end{block}

\pause
\vspace{0.5cm}

\begin{block}{Teorema}
    O movimento Browniano fracionário é uma martingale se e só se $H=\frac{1}{2}$
\end{block}
    
\end{frame}


\begin{frame}{Semi-martingale}

\begin{block}{Definição}
    Um processo $(X_{t})_{t \geq 0}$ diz-se uma semi-martingale se poder ser decomposto como 
    $$X_t=M_t+A_t,$$

    onde $(M_t)_{t \geq 0}$ é uma martingale local e $(A_t)_{t \geq 0}$ é um processo de Càdlàg de variação limitada.
\end{block}

\pause
\vspace{0.5cm}

\begin{block}{Teorema}
    O movimento Browniano fracionário é uma semi-martingale se e só se $H=\frac{1}{2}$
\end{block}

\end{frame}



\begin{frame}{Opções financeiras}
Uma opção é um contracto que concede ao titular o direito, mas não a obrigação, de comprar ou vender uma determinada quantidade de um ativo por um preço de exercício previamente acordado, numa data de vencimento específico.

\vspace{0.5cm}

$K \equiv $ preço de exercício

$T \equiv$ maturidade

\vspace{0.5cm}

Opção Europeia: exercício exatamente no instante $T$

Opção Americana: exercício permitido até ao instante $T$

\end{frame}

\begin{frame}{Preço de opção europeia}
    \begin{block}{Teorema}
        O preço de uma opção de compra Europeia de strike price $K$ e maturidade T, no instante t, é dada por:

        $$C(t,S(t))=S(t)N(d_+)-Ke^{-r(T-t)}N(d_-),$$
        onde 
        $$d_{\pm}=\frac{\ln(S(t)/K)+r(T-t) \pm \frac{1}{2} \sigma^2(T^{2H}-t^{2H})}{\sigma\sqrt{T^{2H}-t^{2H}}}$$
    \end{block}
\end{frame}
